# The invocation arg/env placeholder grammar the runtime substitutes. Bare forms
# resolve to the invocation's own primary endpoint (services) or the primary
# dependency (tasks). The named form ${port:<name>} resolves <name> first against
# the owning service's endpoint ids, then against a service declared via
# connectsTo (services) or requires (tasks) — its primary slot-plan endpoint.
substitution: ${port} ${host} ${stateDir} ${port:<name>} ${host:<name>} ${secret:<id>}
