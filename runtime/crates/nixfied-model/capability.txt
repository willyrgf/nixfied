# The exec arg/env placeholder grammar the runtime substitutes. Bare forms
# resolve to the exec's own primary endpoint (services) or the primary dependency
# (tasks). The named form ${port:<name>} resolves <name> first against the exec's
# own endpoint ids (services), then against a service declared via connectsTo
# (services) or dependsOnServicesReady (tasks) — its primary slot-plan endpoint.
substitution: ${port} ${host} ${stateDir} ${port:<name>} ${host:<name>}
